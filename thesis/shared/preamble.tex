% -- Section depth ------------------------------------------------------------
%  \makeatletter is required: these names contain @, and this is a .tex file

\usepackage{titlesec}
%  Keep a section on one page when it can fit on one (author's choice, 23 Sep).
%  Rule, at each \section: if the section is no taller than a text page but
%  taller than the space left on the current page, start it on a new page;
%  otherwise let it flow. Sections longer than a page flow as usual. The first
%  section of a chapter is exempt, or the chapter heading could stand alone.
%
%  TeX cannot know a section's height before typesetting it, so it is measured:
%  zref-savepos records where each section starts and ends, and the NEXT run
%  reads that height back. latexmk's reruns make this converge on their own.
%  Height excludes the break itself, so the decision does not feed back into
%  the measurement. A float that drifts out of its section is not counted.
\usepackage{zref-savepos,zref-abspage}
\makeatletter
\zref@addprop{savepos}{abspage}
\newcounter{keepsec}
\newif\ifkeepsec@open
\newdimen\keepsec@len \newdimen\keepsec@top \newdimen\keepsec@bot
\newcount\keepsec@ps \newcount\keepsec@pe
\def\keepsec@close{\ifkeepsec@open\par\zsavepos{ks-e-\the\c@keepsec}\global\keepsec@openfalse\fi}
%  Height of section #1 from the previous run; 0pt if not yet measured.
\def\keepsec@measure#1{%
  \keepsec@len=\z@
  \zref@ifrefundefined{ks-s-#1}{}{\zref@ifrefundefined{ks-e-#1}{}{%
    \keepsec@top=\paperheight
    \advance\keepsec@top by -1in \advance\keepsec@top by -\voffset
    \advance\keepsec@top by -\topmargin \advance\keepsec@top by -\headheight
    \advance\keepsec@top by -\headsep
    \keepsec@bot=\keepsec@top \advance\keepsec@bot by -\textheight
    \keepsec@ps=\zref@extractdefault{ks-s-#1}{abspage}{0}\relax
    \keepsec@pe=\zref@extractdefault{ks-e-#1}{abspage}{0}\relax
    \ifnum\keepsec@ps=\keepsec@pe
      \keepsec@len=\zposy{ks-s-#1}sp \advance\keepsec@len by -\zposy{ks-e-#1}sp
    \else
      \keepsec@len=\zposy{ks-s-#1}sp \advance\keepsec@len by -\keepsec@bot
      \advance\keepsec@len by \keepsec@top \advance\keepsec@len by -\zposy{ks-e-#1}sp
      \advance\keepsec@pe by -\keepsec@ps \advance\keepsec@pe by \m@ne
      \advance\keepsec@len by \keepsec@pe\textheight
    \fi}}}
\newcommand{\sectionbreak}{%
  \keepsec@close
  \stepcounter{keepsec}%
  \ifnum\value{section}>0
    \keepsec@measure{\the\c@keepsec}%
    \ifdim\pagegoal<\maxdimen
      \ifdim\keepsec@len>\z@\ifdim\keepsec@len<\textheight
        \ifdim\keepsec@len>\dimexpr\pagegoal-\pagetotal\relax\clearpage\fi
      \fi\fi
    \fi
  \fi
  \zsavepos{ks-s-\the\c@keepsec}\global\keepsec@opentrue}
\AtBeginDocument{\pretocmd{\chapter}{\keepsec@close}{}{}}
\AtEndDocument{\keepsec@close}

%  secnumdepth 3, NOT 4. The chapters use \paragraph{...} as an unnumbered
%  lead-in ("Checkpoint selection."), which is the standard idiom. At depth 4
%  those get numbered, and because the chapters have no subsections in between
%  the numbers come out as "3.3.0.0.3 Checkpoint selection." The ESI templates
%  set 4; that is what produces those. tocdepth follows so the contents list
%  stops at subsubsection instead of listing every lead-in paragraph.

% -- Languages (LOADED LATE ON PURPOSE) --------------------------------------
%  polyglossia pulls in `bidi` as soon as an RTL language is declared, and bidi
%  REQUIRES that it be loaded after amsmath, amstext, caption, xcolor, colortbl
%  and friends. Declaring Arabic near the top of the preamble -- which is where
%  the ESI templates put it -- makes a cold build emit 17 "you have loaded
%  package X after bidi" errors. Keep this block below every package above.
%  babel is deliberately NOT loaded: polyglossia replaces it, and loading both
%  errors under XeLaTeX/LuaLaTeX. The ESI templates load both.
\usepackage{polyglossia}
\setdefaultlanguage{english}

%  Line breaking. The ESI templates set \hyphenpenalty=10000 with \tolerance=1 and
%  \emergencystretch=\maxdimen: hyphenation off, and TeX told to stretch spaces
%  without limit instead -- which silences every warning and prints rivers of
%  white space in justified text. Hyphenate normally, allow a bounded stretch as a
%  last resort, and let microtype hang punctuation into the margin (XeLaTeX
%  supports protrusion only, not font expansion).
\usepackage[protrusion=true,expansion=false]{microtype}
\setlength{\emergencystretch}{1.5em}
\setotherlanguage[numerals=maghrib]{arabic}   % Latin digits inside Arabic text

%  Pick whichever Arabic font this machine actually has, rather than failing the
%  whole build on a missing font. If neither is installed the document still
%  compiles and the Arabic abstract renders in the default face (visibly wrong,
%  which is the point -- it tells you to install one).
\IfFontExistsTF{Amiri}{%
  \newfontfamily\arabicfont[Script=Arabic]{Amiri}%
}{%
  \IfFontExistsTF{Noto Sans Arabic}{%
    \newfontfamily\arabicfont[Script=Arabic]{Noto Sans Arabic}%
  }{%
    \GenericWarning{}{THESIS: no Arabic font found (tried Amiri, Noto Sans
      Arabic). The Arabic abstract will NOT render correctly. Install one
      before the final build: `sudo apt install fonts-hosny-amiri` or
      `fonts-noto-core`.}%
    %  Keep \arabicfont defined so the build still finishes: fall back to the
    %  roman family rather than naming a system font that may also be absent.
    \providecommand{\arabicfont}{\rmfamily}%
  }%
}
